class Solution(object):
    def longestMonotonicSubarray(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        count1 = 1
        count2 = 1
        maxi = 1
        for i in range(1,len(nums)):
            if nums[i-1] < nums[i]:
                count1 += 1
            else:
                count1 = 1
            if nums[i-1] > nums[i]:
                count2 += 1
            else:
                count2 = 1
            maxi =  max(maxi,count1,count2)
        return maxi
