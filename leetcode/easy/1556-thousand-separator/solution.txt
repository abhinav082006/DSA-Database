class Solution(object):
    def thousandSeparator(self, n):
        """
        :type n: int
        :rtype: str
        """
        n = str(n)
        res = ""
        count = 0
        for i in range(len(n)-1 , -1 ,-1):
            res =  n[i] + res
            count += 1
            if count == 3 and i != 0:
                res = '.' + res
                count = 0
        return res
