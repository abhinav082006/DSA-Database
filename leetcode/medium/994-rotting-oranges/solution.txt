        fresh += 1
        minutes = 0
        directions = [(0,1),(0,-1),(-1,0),(1,0)]
        while q and fresh > 0:
            for _ in range(len(q)):
                x,y = q.popleft()
                for dx , dy in directions:
                    nx = x + dx
                    ny = y + dy
                    if 0 <= nx < m and 0 <= ny < n and grid[nx][ny] == 1:
                        grid[nx][ny] = 2
                        fresh -= 1
                        q.append((nx,ny))
            minutes += 1
        return minutes if fresh == 0 else -1
