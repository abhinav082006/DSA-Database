class Solution(object):
    def checkValidString(self, s):
        """
        :type s: str
        :rtype: bool
        """
        low = 0
        high = 0
        for x in s:
            if x == "(":
                low += 1
                high += 1
            if x == ")":
                low -= 1
                high -= 1
            if x == "*":
                low -= 1
                high += 1
            if high < 0:
                return False
            low = max(low,0)
        return low == 0
