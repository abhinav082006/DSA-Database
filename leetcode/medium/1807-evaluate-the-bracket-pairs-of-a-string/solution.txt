class Solution(object):
    def evaluate(self, s, knowledge):
        """
        :type s: str
        :type knowledge: List[List[str]]
        :rtype: str
        """
        knowledge = dict(knowledge)
        while "(" in s:
            st = s.index('(')
            en = s.index(')')
            key = s[st+1:en]
            if key in knowledge:
                value = knowledge[key]
            else:
                value = '?'
            
            s = s[:st] + value + s[en+1:]
        return s
