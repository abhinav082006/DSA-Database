   while low < high:
            mid = (low + high) // 2
            ops = sum(max(0, x - mid) for x in diff)

            if ops <= k:
                high = mid
            else:
                low = mid + 1

        ans = sum(min(x, low) ** 2 for x in diff)
        remaining = k - sum(max(0, x - low) for x in diff)

        for x in diff:
            if remaining > 0 and x >= low and low > 0:
                ans -= low * low - (low - 1) ** 2
                remaining -= 1

        return ans
