class Solution(object):
    def maxProduct(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        n = len(nums)

        dp = [[0] * 2 for _ in range(n)]

        dp[0][0] = nums[0]
        dp[0][1] = nums[0]

        for i in range(1, n):
            x = nums[i]

            dp[i][0] = max(x, dp[i-1][0] * x, dp[i-1][1] * x)
            dp[i][1] = min(x, dp[i-1][0] * x, dp[i-1][1] * x)

        return max(dp[i][0] for i in range(n))
