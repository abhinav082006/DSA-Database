class Solution(object):
    def removeInvalidParentheses(self, s):
        """
        :type s: str
        :rtype: List[str]
        """
        leftRemove = 0
        rightRemove = 0

        for ch in s:
            if ch == '(':
                leftRemove += 1
            elif ch == ')':
                if leftRemove > 0:
                    leftRemove -= 1
                else:
                    rightRemove += 1

        ans = set()

        def backtrack(ind, left, right, open, path):
            if ind == len(s):
                if left == 0 and right == 0 and open == 0:
                    ans.add("".join(path))
